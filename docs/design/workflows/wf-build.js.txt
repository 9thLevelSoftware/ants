const ROOT = '/home/user/foundress'

if (!tasks.length) throw new Error('no tasks given')

const IMPL_SCHEMA = {
  type: 'object',
  properties: {
    summary: { type: 'string' },
    filesTouched: { type: 'array', items: { type: 'string' } },
    contractChanges: { type: 'array', items: { type: 'string' }, description: 'additive changes made to shared contracts or core' },
    verification: { type: 'array', items: { type: 'string' }, description: 'commands run and their results, screenshots looked at' },
    acceptance: {
      type: 'array',
      items: {
        type: 'object',
        properties: { criterion: { type: 'string' }, status: { type: 'string', enum: ['met', 'partial', 'not-met'] }, evidence: { type: 'string' } },
        required: ['criterion', 'status'],
      },
    },
    notDone: { type: 'array', items: { type: 'string' }, description: 'anything in the spec that is not finished, honestly' },
    needsFromOthers: { type: 'array', items: { type: 'string' } },
    notesForIntegrator: { type: 'array', items: { type: 'string' } },
  },
  required: ['summary', 'filesTouched', 'verification', 'acceptance', 'notDone'],
}

const VERIFY_SCHEMA = {
  type: 'object',
  properties: {
    verdict: { type: 'string', enum: ['pass', 'pass-with-minor', 'fail'] },
    checksRun: { type: 'array', items: { type: 'string' } },
    findings: {
      type: 'array',
      items: {
        type: 'object',
        properties: {
          severity: { type: 'string', enum: ['blocker', 'major', 'minor'] },
          title: { type: 'string' },
          where: { type: 'string', description: 'file:line or command' },
          detail: { type: 'string', description: 'what is wrong, how to reproduce, what the spec says' },
        },
        required: ['severity', 'title', 'detail'],
      },
    },
    testsAdded: { type: 'array', items: { type: 'string' } },
  },
  required: ['verdict', 'checksRun', 'findings'],
}

const implPrompt = (t) => COMMON + `YOUR TASK SPEC: ${ROOT}/${t.spec}
Read it now and implement it completely. Work through its acceptance criteria one by one and report each honestly (met / partial / not-met with evidence).`

const verifyPrompt = (t, impl) => COMMON + `YOU ARE NOT THE IMPLEMENTER. You are an independent, adversarial reviewer and tester.
Another engineer has just implemented the task spec ${ROOT}/${t.spec}. Their report (treat every claim as unproven):
${JSON.stringify(impl).slice(0, 16000)}

Your job is to find what is wrong or missing before it costs the team time later:
1. Read the task spec and the documents it lists, then read ALL the code in the task's owned paths.
2. Run "npm run typecheck" and the task's tests yourself. ${t.visual ? 'Run the harness for the visual acceptance criteria, take your own screenshots and LOOK at them.' : ''}
3. Check every acceptance criterion against the code and by running it. Hunt for: spec violations, logic and numerical bugs, edge cases (zero, negative, NaN, empty, huge), nondeterminism, layering violations (three/DOM/Math.random in sim code), state that is not plain serialisable data, magic numbers that belong in balance files, performance traps in hot paths (per-tick allocations, O(n^2) scans), missing or weak tests, files touched outside the owned paths.
4. Write additional tests that try to break the implementation, inside the task's owned paths (name them *.verify.test.ts). Keep tests that pass as regression tests; a failing test that demonstrates a real bug stays in place for the fixer.
5. Do NOT modify production code. Report findings with severity (blocker = wrong/unsafe to build on; major = spec criterion not met or real bug; minor = quality), a precise location and a reproduction.
Be concrete and fair: no style nitpicks, no speculative findings without evidence. If it is good, say pass.`

const fixPrompt = (t, impl, verify) => COMMON + `You are the engineer responsible for the task spec ${ROOT}/${t.spec} (already implemented; summary: ${String(impl && impl.summary).slice(0, 1500)}).
An independent reviewer reported these findings:
${JSON.stringify(verify.findings).slice(0, 80000)}
Tests the reviewer added: ${JSON.stringify(verify.testsAdded || [])}

For each finding: reproduce it; if it is real, fix it properly (root cause, not the symptom) and make sure a test covers it; if it is not real, say why with evidence. Then re-run typecheck and all tests in your paths${t.visual ? ' and the harness screenshots (look at them)' : ''}. Do not weaken or delete the reviewer's tests unless a test itself is wrong (explain). Report what you fixed and anything still open in notDone.`

const results = await pipeline(
  tasks,
  (t) => {
    // skipImpl: the task was already implemented in an earlier run; go straight to independent verification.
    if (t.startAt === 'fix') return { summary: 'Implemented and reviewed in an earlier run. The implementer report and the full review are in ' + ROOT + '/' + t.reviewPath + ' (read it first).', filesTouched: [], verification: [], acceptance: [], notDone: [] }
    if (t.skipImpl) return { summary: t.implSummary || 'Implemented in an earlier run; no report available — judge the code itself.', filesTouched: [], verification: [], acceptance: [], notDone: [] }
    return agent(implPrompt(t), { label: 'impl:' + t.key, phase: 'Implement', schema: IMPL_SCHEMA, model: t.model || 'opus', effort: t.effort || 'high' })
  },
  async (impl, t) => {
    if (!impl) return { impl: null, verify: null }
    if (t.verify === false) return { impl, verify: null }
    if (t.startAt === 'fix') return { impl, verify: { verdict: 'fail', checksRun: [], testsAdded: [], findings: [{ severity: 'major', title: 'Full review in a file', detail: 'The complete list of review findings (with locations, reproductions and the tests the reviewer added) is the "review" object in ' + ROOT + '/' + t.reviewPath + '; the implementer\'s own report is the "implReport" object there. Read the whole file and treat every finding in it as if it were listed here. An earlier fix engineer was interrupted part-way: some findings may already be fixed or half-fixed in the tree, so check the current state of each before changing code, and report on all of them.' }] } }
    const verify = await agent(verifyPrompt(t, impl), { label: 'verify:' + t.key, phase: 'Verify', schema: VERIFY_SCHEMA, model: t.verifyModel || 'opus', effort: 'high' })
    return { impl, verify }
  },
  async (r, t) => {
    if (!r || !r.impl || !r.verify) return r
    const real = (r.verify.findings || []).filter((f) => f.severity === 'blocker' || f.severity === 'major')
    if (!real.length) return { ...r, fix: null }
    const fix = await agent(fixPrompt(t, r.impl, r.verify), { label: 'fix:' + t.key, phase: 'Fix', schema: IMPL_SCHEMA, model: t.model || 'opus', effort: t.effort || 'high' })
    return { ...r, fix }
  },
)

return tasks.map((t, i) => {
  const r = results[i]
  if (!r) return { key: t.key, failed: true }
  const last = r.fix || r.impl
  return {
    key: t.key,
    summary: last && last.summary,
    notDone: last && last.notDone,
    acceptanceNotMet: ((last && last.acceptance) || []).filter((a) => a.status !== 'met'),
    contractChanges: [].concat((r.impl && r.impl.contractChanges) || [], (r.fix && r.fix.contractChanges) || []),
    needsFromOthers: [].concat((r.impl && r.impl.needsFromOthers) || [], (r.fix && r.fix.needsFromOthers) || []),
    notesForIntegrator: [].concat((r.impl && r.impl.notesForIntegrator) || [], (r.fix && r.fix.notesForIntegrator) || []),
    verifyVerdict: r.verify && r.verify.verdict,
    findings: ((r.verify && r.verify.findings) || []).map((f) => f.severity + ': ' + f.title),
    fixed: !!r.fix,
  }
})
