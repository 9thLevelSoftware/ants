const ROUND_SCHEMA = {

    triedAndReverted: { type: 'array', items: { type: 'string' } },
    gddDeviations: { type: 'array', items: { type: 'string' } },

const roundPrompt = (n, prev, withNote) => COMMON + `YOU ARE THE ENGINEER OF ROUND ${n} of the colony-life task.
${prev ? 'The independent sweep and review after the previous round reported (also recorded in the ledger):\n' + JSON.stringify(prev).slice(0, 30000) + '\nFix every blocker and major finding in it first (reproduce each; if one is not real, say why with evidence), then continue with the method.' : 'No independent sweep has been run for this task yet: your baseline is the sweep table named in the spec.'}
Follow the spec's method exactly: account first, rank causes by measured loss, fix the top one to three at the cause with tests and a before/after on the affected runs, append "## Round ${n}" to the ledger. Do not run the full 15-run sweep. Evidence under test-output/colony-life/round-${n}/.` + (args && args.note && withNote ? '\n\nNOTE FROM THE INTEGRATOR: ' + args.note : '') + progress(n, 'round')

const sweepPrompt = (n, round) => COMMON + `YOU ARE NOT THE ENGINEER OF THE ROUND. You are the independent engineer who measures and reviews round ${n} of the colony-life task.
The round's engineer reported (treat every claim as unproven):
${JSON.stringify(round).slice(0, 20000)}

Do this, in order:
1. Run the full sweep on the current tree: seeds 1-3 x woodland-edge, suburban-garden, playground-park, car-park-verge, kitchen-door with "node tools/sim/run-ts.mjs tools/sim/life.ts --seed S --archetype A --until winter --cache test-output/viab/cache --out test-output/colony-life/round-${n}/sweep" (Naturalist, default start, flooding on, no player; at most 2 runs at a time; test-output/econ-growth/sweep.sh and table.py show how the last sweep and its table were made — reuse or adapt them under tools/sim if useful). Write the table to test-output/colony-life/round-${n}/sweep-table.md with the same columns as test-output/econ-growth-review/sweep9-table.md plus first-worker count and worker:lost.
1b. Year two (spec directions 6 and 14, rulings R22 and R24; never skip it): after the 15 runs, from the same frozen copy of the tree, run the five long runs "PAR=2 UNTIL_DAY=32 tools/sim/longrun-life.sh <abs out test-output/colony-life/round-${n}/sweep-long> <abs cache test-output/viab/cache> \"woodland-edge:2 suburban-garden:2 car-park-verge:1 playground-park:3 kitchen-door:3\"" (skip any whose JSON already exists there: the integrator may have run them on the same tree hash; check the hash), then tools/sim/longrun-table.py. Report R22's two year-two checks and R24 by number.
2. Compute both bars from the table by script, not by eye: the viability bar and the growth bar R19 exactly as written in the spec. Compare every run with the previous sweep table (the ledger names it; the first is test-output/econ-growth-review/sweep9-table.md) and list regressions with numbers.
3. While the sweep runs (it takes over an hour), review the round's code changes ("git diff" against the last commit, plus what the ledger's round section lists): does each fix address its cause rather than a symptom; is anything forbidden by the spec's "What is not allowed" (lifespans, needs, free food, weakened bars or ruling tests, special cases for a seed or archetype); determinism (persisted triggers, memo keys, no wall clock); per-tick cost traps; tests that would not fail without the fix. Run typecheck and the tests of the folders the round touched, folder by folder. Write *.verify.test.ts tests that try to break the fixes; a failing test that shows a real bug stays in place. Do NOT modify production code.
4. Append "### Independent sweep after round ${n}" to the ledger ${LEDGER}: the table path, the bar numbers, regressions, findings, and the leads you see in the failing runs' own diagnostics for the next round (most costly first).
Be concrete and fair: no style nitpicks, no speculative findings without evidence.` + progress(n, 'sweep')

const history = []
let prev = null
let noteGiven = false
for (let n = FIRST; n < FIRST + MAX; n++) {
  const given = n === FIRST && args && args.firstRoundResult
  const round = given ? args.firstRoundResult : await agent(roundPrompt(n, prev, !noteGiven), { label: 'round:' + n, phase: 'Round', schema: ROUND_SCHEMA, model: MODEL, effort: EFFORT })
  if (!given) noteGiven = true
  if (!round) { log('round ' + n + ' returned nothing; stopping'); history.push({ round: n, failed: true }); break }
  const sweep = await agent(sweepPrompt(n, round), { label: 'sweep:' + n, phase: 'Sweep', schema: SWEEP_SCHEMA, model: MODEL, effort: 'high' })
  history.push({ round: n, summary: round.summary, causes: round.causesRanked, fixes: (round.fixes || []).map((f) => f.cause + ' :: ' + f.beforeAfter), deviations: round.gddDeviations, notes: round.notesForIntegrator, sweep })
  if (!sweep) { log('sweep ' + n + ' returned nothing; stopping'); break }
  const serious = (sweep.findings || []).filter((f) => f.severity === 'blocker' || f.severity === 'major')
  log('round ' + n + ': viability ' + sweep.viabilityMet + ', growth ' + sweep.growthMet + ', serious findings ' + serious.length + ' — ' + JSON.stringify(sweep.numbers))
  if (sweep.viabilityMet && sweep.growthMet && !serious.length) break
  prev = sweep
}
if (history.length === MAX && !(history[MAX - 1].sweep && history[MAX - 1].sweep.growthMet)) log('stopped at the round cap of ' + MAX + ' without the growth bar holding')
return history
